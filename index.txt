<script>
(function(){
  const URL_ = (window.SCRIPT_URL||'').trim();
  const hasBackend = /^https:\/\/script\.google(usercontent)?\.com\//.test(URL_);
  if(!hasBackend) document.getElementById('cfgBanner').style.display='block';

  const $ = id => document.getElementById(id);
  const RES = ['Residing','Not Residing'], F6 = ['Yes','No'];
  let blos = [], cur = null, rows = [], ans = {}, saved = {}, dirty = new Set(), busy = false;

  const store = {
    get(k){ try{ return JSON.parse(localStorage.getItem(k)); }catch(e){ return null; } },
    set(k,v){ try{ localStorage.setItem(k, JSON.stringify(v)); }catch(e){} },
    del(k){ try{ localStorage.removeItem(k); }catch(e){} }
  };
  const esc = s => String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
  const tagCls = s => s==='Permanently Shifted'?'t-ps':s==='Untraceable/Absent'?'t-ua':s==='Death'?'t-d':'t-ae';
  const opts = (list, v) => '<option value="">-- Select --</option>' + list.map(o => `<option${o===v?' selected':''}>${o}</option>`).join('');
  const setMsg = (t, cls) => { const m=$('msg'); m.textContent=t; m.className=cls||''; };

  async function getJSON(u){ const r = await fetch(u, {cache:'no-store'}); if(!r.ok) throw new Error('HTTP '+r.status); return r.json(); }

  async function init(){
    blos = await getJSON('data/blos.json');
    $('bloSel').insertAdjacentHTML('beforeend', blos.map(b =>
      `<option value="${b.ps}">PS-${b.ps} – ${esc(b.blo)} (${esc(b.part)})</option>`).join(''));
    const fromUrl = new URLSearchParams(location.search).get('ps');
    const last = fromUrl || store.get('asdd_last_ps');
    if(last && blos.some(b => String(b.ps)===String(last))){ $('bloSel').value = String(last); loadPS(); }
  }

  async function loadPS(){
    if(dirty.size && !confirm('You have unsaved changes. Switch BLO anyway? (They stay saved on this device as a draft.)')){
      $('bloSel').value = cur ? String(cur.ps) : ''; return;
    }
    const ps = $('bloSel').value;
    if(!ps){ $('work').classList.add('hide'); $('savebar').style.display='none'; $('partMeta').textContent=''; cur=null; return; }
    cur = blos.find(b => String(b.ps)===ps);
    store.set('asdd_last_ps', ps);
    $('partMeta').textContent = `Part No. ${cur.ps} – ${cur.part} · BLO: ${cur.blo} · ${cur.n} electors in ASDD list`;
    $('work').classList.remove('hide'); $('savebar').style.display='block';
    $('tb').innerHTML=''; $('loader').classList.remove('hide'); $('none').classList.add('hide');
    setMsg('Loading…');
    try{
      const [data, sv] = await Promise.all([
        getJSON(`data/ps/${ps}.json`),
        hasBackend ? getJSON(`${URL_}?action=get&ps=${ps}`).catch(e => ({ok:false,error:e.message})) : Promise.resolve({ok:true,data:{}})
      ]);
      if(cur.ps !== Number(ps)) return;
      rows = data;
      saved = (sv && sv.ok && sv.data) ? sv.data : {};
      ans = {}; dirty = new Set();
      for(const r of rows){ const s = saved[r[1]]; ans[r[1]] = s ? [s[0]||'', s[1]||''] : ['','']; }
      // restore unsaved draft from this device
      const draft = store.get('asdd_draft_'+ps) || {};
      for(const ep in draft){ if(ans[ep]){ ans[ep] = draft[ep]; if(isChanged(ep)) dirty.add(ep); } }
      if(sv && sv.ok===false) setMsg('Could not read saved data from Google Sheet — check internet and tap "Reload from sheet".','err');
      else if(dirty.size) setMsg(`${dirty.size} unsaved change(s) restored from this device. Tap Submit to save.`,'warn');
      else setMsg(Object.keys(saved).length ? 'Saved answers loaded from Google Sheet.' : 'Fill both dropdowns for each elector, then tap Submit.');
      render(); updateStats();
    }catch(e){ setMsg('Failed to load list: '+e.message,'err'); }
    finally{ $('loader').classList.add('hide'); }
  }

  function isChanged(ep){ const s = saved[ep] || ['','']; const a = ans[ep]; return (a[0]||'')!==(s[0]||'') || (a[1]||'')!==(s[1]||''); }

  function render(){
    const q = $('q').value.trim().toUpperCase(), fs = $('fStatus').value, fw = $('fShow').value;
    let html = '', n = 0;
    for(const r of rows){
      const [sn, epic, name, rt, rn, op, os, , , st] = r;
      const a = ans[epic], done = a[0] && a[1];
      if(fs && st!==fs) continue;
      if(fw==='pending' && done) continue;
      if(fw==='done' && !done) continue;
      if(q && !(name.toUpperCase().includes(q) || epic.includes(q) || String(sn)===q || String(os)===q)) continue;
      n++;
      html += `<tr data-ep="${epic}" class="${done?'done':''}${dirty.has(epic)?' dirty':''}">
        <td class="c-sn">${sn}</td><td class="c-dist">CENTRAL</td><td class="c-acno">24</td><td class="c-acname">PATEL NAGAR</td>
        <td class="c-name">${esc(name)}</td><td><span class="lbl-m">EPIC No.</span>${epic}</td>
        <td><span class="lbl-m">Relation</span>${esc(rt)}</td><td class="c-rel-name"><span class="lbl-m">Relation Name</span>${esc(rn)}</td>
        <td class="c-part">${esc(op)}</td><td><span class="lbl-m">Serial No</span>${esc(os)}</td>
        <td class="c-rel"><span class="lbl-m">Status in ASDD list</span><span class="tag ${tagCls(st)}">${esc(st)}</span></td>
        <td class="c-sel"><span class="lbl-m">After re-verification</span><select data-k="0" class="${a[0]?'':'empty'}">${opts(RES,a[0])}</select></td>
        <td class="c-sel"><span class="lbl-m">Form-6 obtained?</span><select data-k="1" class="${a[1]?'':'empty'}">${opts(F6,a[1])}</select></td>
      </tr>`;
    }
    $('tb').innerHTML = html;
    $('none').classList.toggle('hide', n>0);
    $('shown').textContent = `Showing ${n} of ${rows.length}`;
  }

  function updateStats(){
    let done=0,res=0,f6=0;
    for(const r of rows){ const a=ans[r[1]]; if(a[0]&&a[1]) done++; if(a[0]==='Residing') res++; if(a[1]==='Yes') f6++; }
    $('sTotal').textContent = rows.length; $('sDone').textContent = done; $('sPend').textContent = rows.length-done;
    $('sRes').textContent = res; $('sF6').textContent = f6;
    $('sBar').style.width = (rows.length ? done/rows.length*100 : 0)+'%';
    $('save').disabled = busy || !dirty.size || !hasBackend;
    if(!busy && dirty.size) setMsg(`${dirty.size} unsaved change(s). Tap Submit to save to Google Sheet.`,'warn');
  }

  function saveDraft(){
    const d = {}; dirty.forEach(ep => d[ep] = ans[ep]);
    if(dirty.size) store.set('asdd_draft_'+cur.ps, d); else store.del('asdd_draft_'+cur.ps);
  }

  $('tb').addEventListener('change', e => {
    const sel = e.target; if(sel.tagName!=='SELECT') return;
    const tr = sel.closest('tr'), ep = tr.dataset.ep, k = +sel.dataset.k;
    ans[ep][k] = sel.value;
    sel.classList.toggle('empty', !sel.value);
    if(isChanged(ep)) dirty.add(ep); else dirty.delete(ep);
    tr.classList.toggle('dirty', dirty.has(ep));
    tr.classList.toggle('done', !!(ans[ep][0] && ans[ep][1]));
    saveDraft(); updateStats();
  });

  async function submit(){
    if(!dirty.size || busy) return;
    const list = rows.filter(r => dirty.has(r[1]));
    const half = list.filter(r => { const a=ans[r[1]]; return !!a[0] !== !!a[1]; }).length;
    if(half && !confirm(`${half} elector(s) have only one of the two dropdowns filled. Submit anyway?`)) return;
    busy = true; $('save').disabled = true; $('save').textContent = 'Saving…';
    setMsg(`Saving ${list.length} elector(s)…`);
    const payload = { ps: cur.ps, blo: cur.blo, partName: cur.part,
      rows: list.map(r => [r[0], r[1], r[2], r[3], r[4], r[5], r[6], r[7], r[8], r[9], ans[r[1]][0], ans[r[1]][1]]) };
    try{
      const res = await fetch(URL_, { method:'POST', body: JSON.stringify(payload) });
      const out = await res.json();
      if(!out.ok) throw new Error(out.error || 'Server error');
      list.forEach(r => { saved[r[1]] = ans[r[1]].slice(); });
      dirty.clear(); saveDraft();
      document.querySelectorAll('#tb tr.dirty').forEach(tr => tr.classList.remove('dirty'));
      busy = false; updateStats();
      setMsg(`Saved ✓ ${list.length} elector(s) to Google Sheet at ${new Date().toLocaleTimeString()}.`,'ok');
    }catch(e){
      busy = false; updateStats();
      setMsg('Not saved: '+e.message+'. Your selections are kept on this device — try Submit again.','err');
    }finally{ $('save').textContent = 'Submit'; }
  }

  $('bloSel').addEventListener('change', loadPS);
  $('save').addEventListener('click', submit);
  $('reload').addEventListener('click', () => {
    if(dirty.size && !confirm('Discard unsaved changes and reload saved data from the sheet?')) return;
    if(cur){ store.del('asdd_draft_'+cur.ps); dirty.clear(); loadPS(); }
  });
  let t; $('q').addEventListener('input', () => { clearTimeout(t); t = setTimeout(render, 150); });
  $('fStatus').addEventListener('change', render); $('fShow').addEventListener('change', render);
  window.addEventListener('beforeunload', e => { if(dirty.size){ e.preventDefault(); e.returnValue=''; } });
  /* ---------- Admin ---------- */
  const ADMIN_USER = 'admin', ADMIN_PASS = '2424';
  let admList = [], admSort = 'ps', admAsc = true;
  const ss = { get(k){ try{ return sessionStorage.getItem(k); }catch(e){ return null; } }, set(k,v){ try{ sessionStorage.setItem(k,v); }catch(e){} }, del(k){ try{ sessionStorage.removeItem(k); }catch(e){} } };

  function showAdmin(on){
    $('adminView').classList.toggle('hide', !on);
    $('selPanel').classList.toggle('hide', on);
    $('work').classList.toggle('hide', on || !cur);
    $('savebar').style.display = (!on && cur) ? 'block' : 'none';
    $('adminBtn').classList.toggle('hide', on);
    if(on) loadAdmin();
    window.scrollTo(0,0);
  }
  $('adminBtn').addEventListener('click', () => {
    if(ss.get('asdd_admin')==='1') return showAdmin(true);
    $('lu').value=''; $('lp').value=''; $('lerr').textContent='';
    $('loginModal').classList.remove('hide'); $('lu').focus();
  });
  $('lcancel').addEventListener('click', () => $('loginModal').classList.add('hide'));
  $('loginForm').addEventListener('submit', e => {
    e.preventDefault();
    if($('lu').value.trim().toLowerCase()===ADMIN_USER && $('lp').value===ADMIN_PASS){
      ss.set('asdd_admin','1'); $('loginModal').classList.add('hide'); showAdmin(true);
    } else $('lerr').textContent = 'Wrong username or password.';
  });
  $('admExit').addEventListener('click', () => { ss.del('asdd_admin'); showAdmin(false); });
  $('admRefresh').addEventListener('click', loadAdmin);

  async function loadAdmin(){
    $('admMsg').textContent = 'Loading…';
    let sum = {};
    if(!hasBackend) $('admMsg').textContent = 'Google Sheet link not set in config.js.';
    else try{
      const r = await getJSON(URL_ + '?action=summary');
      if(!r.ok) throw new Error(r.error); sum = r.data;
      $('admMsg').textContent = 'Updated ' + new Date().toLocaleTimeString();
    }catch(e){ $('admMsg').textContent = 'Could not read Google Sheet: ' + e.message; }
    admList = blos.map(b => { const s = sum[b.ps] || {}, done = s.done||0;
      return {...b, done, pct: b.n ? done/b.n*100 : 0, residing:s.residing||0, notResiding:s.notResiding||0, f6yes:s.f6yes||0, f6no:s.f6no||0}; });
    const T = k => admList.reduce((a,b) => a + b[k], 0), tot = T('n'), dn = T('done');
    $('admStats').innerHTML = [
      ['Total ASDD electors', tot.toLocaleString('en-IN')], ['Completed', `${dn.toLocaleString('en-IN')} (${tot?(dn/tot*100).toFixed(1):0}%)`],
      ['BLOs not started', admList.filter(b => !b.done).length], ['BLOs 100% done', admList.filter(b => b.done >= b.n).length],
      ['Residing', T('residing').toLocaleString('en-IN')], ['Form-6 obtained', T('f6yes').toLocaleString('en-IN')]
    ].map(([k,v]) => `<div class="stat"><div class="k">${k}</div><div class="v">${v}</div></div>`).join('');
    drawAdmin();
  }
  function drawAdmin(){
    const q = $('admQ').value.trim().toUpperCase(), f = $('admF').value;
    const list = admList.filter(b => (!q || String(b.ps)===q || b.blo.toUpperCase().includes(q) || b.part.toUpperCase().includes(q))
      && (!f || (f==='zero' && !b.done) || (f==='part' && b.done && b.done<b.n) || (f==='full' && b.done>=b.n)))
      .sort((a,b) => { const x=a[admSort], y=b[admSort]; return (x>y?1:x<y?-1:0) * (admAsc?1:-1); });
    $('admTb').innerHTML = list.map(b => `<tr class="${b.done?'':'zero'}"><td>${b.ps}</td><td class="l">${esc(b.blo)}</td><td class="l">${esc(b.part)}</td>
      <td>${b.n}</td><td>${b.done}</td><td><span class="pbar"><i style="width:${b.pct}%"></i></span>${b.pct.toFixed(0)}%</td>
      <td>${b.residing}</td><td>${b.notResiding}</td><td>${b.f6yes}</td><td>${b.f6no}</td></tr>`).join('');
  }
  document.querySelectorAll('#admTable th').forEach(th => th.addEventListener('click', () => {
    const k = th.dataset.s; admAsc = admSort===k ? !admAsc : true; admSort = k; drawAdmin(); }));
  $('admQ').addEventListener('input', drawAdmin); $('admF').addEventListener('change', drawAdmin);

  init().catch(e => setMsg('Failed to start: '+e.message,'err'));
})();
